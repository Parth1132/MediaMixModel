import numpy as np
from scipy.optimize import minimize, nnls

# 1. Geometric Adstock Decay
def geometric_adstock(x, alpha, l_max=8):
    weights = np.array([alpha ** l for l in range(l_max)])
    weights = weights / np.sum(weights)
    return np.convolve(x, weights, mode='full')[:len(x)]

# 2. Hill Saturation Function
def hill_saturation(x, K, S):
    x_safe = np.maximum(x, 1e-6)
    return (x_safe ** S) / (x_safe ** S + K ** S)

# 3. SLSQP Constrained Budget Allocator
def optimize_budget(total_budget, channels, params, betas):
    def neg_revenue(spend):
        return -sum(betas[i] * hill_saturation(spend[i], params[ch]['K'], params[ch]['S'])
                   for i, ch in enumerate(channels))
    
    constraints = [{'type': 'eq', 'fun': lambda x: sum(x) - total_budget}]
    bounds = [(total_budget * 0.15, total_budget * 0.55) for _ in channels]
    return minimize(neg_revenue, x0=[total_budget/3]*3, bounds=bounds, constraints=constraints, method='SLSQP')
